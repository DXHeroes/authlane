## Overview

This guide will walk you through:

1. Installing the Authlane SDK
2. Setting up a tenant
3. Configuring a GitHub OAuth app
4. Connecting to GitHub
5. Using the connection credentials

**Time to complete:** ~5 minutes

## Prerequisites

<AccordionGroup>
  <Accordion title="Running Authlane Instance">
    You need either:
    - **Cloud**: Access to `https://api.authlane.com` (coming soon)
    - **Self-hosted**: Follow [Self-Hosting Guide](/guides/self-hosting) first

    For this quickstart, we'll assume you have a local instance running at `http://localhost:3000`
  </Accordion>

  <Accordion title="GitHub OAuth App">
    Create a GitHub OAuth app:
    1. Go to [github.com/settings/developers](https://github.com/settings/developers)
    2. Click "New OAuth App"
    3. Set callback URL: `http://localhost:3000/api/v1/oauth/callback`
    4. Save your **Client ID** and **Client Secret**
  </Accordion>
</AccordionGroup>

## Step 2: Initialize Client

Create a new file `authlane-demo.ts`:

```typescript
import { Authlane } from '@authlane/sdk';

const authlane = new Authlane({
  apiKey: process.env.AUTHLANE_API_KEY!, // Get from dashboard
  baseUrl: 'http://localhost:3000', // or https://api.authlane.com
});

async function main() {
  // We'll add code here
}

main();
```

## Step 3: Configure GitHub Service

First, register your GitHub OAuth app credentials with Authlane:

```bash
curl -X POST http://localhost:3000/api/v1/admin/services/github/credentials \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer YOUR_ADMIN_API_KEY" \
  -d '{
    "tenantId": "your-tenant-id",
    "clientId": "YOUR_GITHUB_CLIENT_ID",
    "clientSecret": "YOUR_GITHUB_CLIENT_SECRET"
  }'
```

<Note>
  **Admin API Key**: Get this from your Authlane dashboard at Settings → API Keys
</Note>

## Step 4: Initiate OAuth Flow

Add to your `main()` function:

```typescript
async function main() {
  const userId = 'user_123'; // Your user's unique ID

  // Step 1: Get OAuth authorization URL
  const { data: authUrl, error } = await authlane.oauth.getAuthorizationUrl({
    userId,
    serviceId: 'github',
    redirectUri: 'http://localhost:3000/oauth/callback', // Your app's callback
  });

  if (error) {
    console.error('Failed to get auth URL:', error);
    return;
  }

  console.log('Visit this URL to authorize:');
  console.log(authUrl);

  // Step 2: User visits URL and approves
  // Step 3: GitHub redirects to your callback with code
  // Step 4: Exchange code for connection

  // (In a real app, this would be in your callback handler)
  const code = 'AUTHORIZATION_CODE_FROM_CALLBACK';
  const state = 'STATE_FROM_CALLBACK';

  const { data: connection, error: connectError } = await authlane.oauth.exchangeCode({
    code,
    state,
  });

  if (connectError) {
    console.error('Failed to connect:', connectError);
    return;
  }

  console.log('Connected successfully!', connection);
}
```

## Step 5: Use Connection Credentials

Once connected, retrieve and use credentials:

```typescript
// Get decrypted credentials
const { data: credentials, error } = await authlane.connections.getCredentials({
  userId: 'user_123',
  serviceId: 'github',
});

if (error) {
  console.error('Failed to get credentials:', error);
  return;
}

// Use credentials with GitHub API
const octokit = new Octokit({
  auth: credentials.access_token,
});

const { data: repos } = await octokit.repos.listForAuthenticatedUser();
console.log('Your repositories:', repos.map(r => r.name));
```

## Step 6: Auto Token Refresh

Authlane automatically refreshes tokens in the background. Check connection health:

```typescript
const { data: health, error } = await authlane.connections.health({
  userId: 'user_123',
  serviceId: 'github',
});

console.log('Connection status:', health.status); // 'connected' | 'expired' | 'error'
console.log('Last refresh:', health.lastRefreshed);
console.log('Next refresh:', health.nextRefresh);
```

## Complete Example

<CodeGroup>

```typescript Express.js
import express from 'express';
import { Authlane } from '@authlane/sdk';

const app = express();
const authlane = new Authlane({
  apiKey: process.env.AUTHLANE_API_KEY!,
  baseUrl: 'http://localhost:3000',
});

// Initiate OAuth
app.get('/connect/github', async (req, res) => {
  const userId = req.user.id; // From your auth middleware

  const { data: authUrl, error } = await authlane.oauth.getAuthorizationUrl({
    userId,
    serviceId: 'github',
    redirectUri: 'http://localhost:4000/oauth/callback',
  });

  if (error) {
    return res.status(500).json({ error: error.message });
  }

  res.redirect(authUrl);
});

// Handle OAuth callback
app.get('/oauth/callback', async (req, res) => {
  const { code, state } = req.query;

  const { data: connection, error } = await authlane.oauth.exchangeCode({
    code: code as string,
    state: state as string,
  });

  if (error) {
    return res.status(500).json({ error: error.message });
  }

  res.json({ message: 'Connected successfully!', connection });
});

// Use connection
app.get('/github/repos', async (req, res) => {
  const userId = req.user.id;

  const { data: credentials, error } = await authlane.connections.getCredentials({
    userId,
    serviceId: 'github',
  });

  if (error) {
    return res.status(500).json({ error: error.message });
  }

  // Use credentials with GitHub API
  const response = await fetch('https://api.github.com/user/repos', {
    headers: {
      Authorization: `Bearer ${credentials.access_token}`,
    },
  });

  const repos = await response.json();
  res.json(repos);
});

app.listen(4000);
```

```typescript Next.js
// app/api/connect/github/route.ts
import { Authlane } from '@authlane/sdk';

const authlane = new Authlane({
  apiKey: process.env.AUTHLANE_API_KEY!,
  baseUrl: process.env.AUTHLANE_API_URL!,
});

export async function GET(request: Request) {
  const session = await getSession(request); // Your auth

  const { data: authUrl, error } = await authlane.oauth.getAuthorizationUrl({
    userId: session.user.id,
    serviceId: 'github',
    redirectUri: `${process.env.APP_URL}/api/oauth/callback`,
  });

  if (error) {
    return Response.json({ error: error.message }, { status: 500 });
  }

  return Response.redirect(authUrl);
}

// app/api/oauth/callback/route.ts
export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);
  const code = searchParams.get('code')!;
  const state = searchParams.get('state')!;

  const { data: connection, error } = await authlane.oauth.exchangeCode({
    code,
    state,
  });

  if (error) {
    return Response.redirect(`/error?message=${error.message}`);
  }

  return Response.redirect('/dashboard?connected=true');
}

// app/api/github/repos/route.ts
export async function GET(request: Request) {
  const session = await getSession(request);

  const { data: credentials, error } = await authlane.connections.getCredentials({
    userId: session.user.id,
    serviceId: 'github',
  });

  if (error) {
    return Response.json({ error: error.message }, { status: 500 });
  }

  const response = await fetch('https://api.github.com/user/repos', {
    headers: {
      Authorization: `Bearer ${credentials.access_token}`,
    },
  });

  const repos = await response.json();
  return Response.json(repos);
}
```

## Troubleshooting

<AccordionGroup>
  <Accordion title="OAuth redirect mismatch error">
    Ensure your callback URL exactly matches what's configured in:
    1. Your OAuth app settings (GitHub/Slack/etc)
    2. Your Authlane service configuration
    3. Your SDK `redirectUri` parameter
  </Accordion>

  <Accordion title="Token refresh failed">
    Check that:
    1. Your OAuth app has `offline_access` scope (or equivalent)
    2. Redis is running (required for BullMQ job queue)
    3. Environment variable `REDIS_URL` is set
  </Accordion>

  <Accordion title="Credentials decryption failed">
    Verify:
    1. `ENCRYPTION_KEY` environment variable is set
    2. The key hasn't changed since credentials were stored
    3. The key is exactly 32 bytes (64 hex characters)
  </Accordion>
</AccordionGroup>
