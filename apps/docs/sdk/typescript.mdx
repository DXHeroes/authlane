## Quick Start

```typescript
import { Authlane } from '@authlane/sdk';

const authlane = new Authlane({
  apiKey: process.env.AUTHLANE_API_KEY!,
  baseUrl: 'https://api.authlane.com', // or your self-hosted URL
});

// List all connections for a user
const { data, error } = await authlane.connections.list({
  userId: 'user_123',
});

if (error) {
  console.error('Error:', error.message);
} else {
  console.log('Connections:', data);
}
```

## Client Configuration

### Authlane Constructor

```typescript
new Authlane(options: AuthlaneOptions)
```

<ParamField path="options.apiKey" type="string" required>
  Your Authlane API key (starts with `sk_`)
</ParamField>

<ParamField path="options.baseUrl" type="string" default="https://api.authlane.com">
  Base URL for Authlane API
</ParamField>

<ParamField path="options.timeout" type="number" default="30000">
  Request timeout in milliseconds
</ParamField>

<ParamField path="options.retries" type="number" default="3">
  Number of retry attempts for failed requests
</ParamField>

### Example

```typescript
const authlane = new Authlane({
  apiKey: process.env.AUTHLANE_API_KEY!,
  baseUrl: 'http://localhost:3000', // Self-hosted
  timeout: 10000, // 10 seconds
  retries: 2,
});
```

## Connections

### List Connections

```typescript
authlane.connections.list(params: ListConnectionsParams): Promise<Response<Connection[]>>
```

<ParamField path="params.userId" type="string" required>
  User ID to list connections for
</ParamField>

<ParamField path="params.serviceId" type="string">
  Filter by specific service (optional)
</ParamField>

```typescript
const { data, error } = await authlane.connections.list({
  userId: 'user_123',
  serviceId: 'github', // optional
});
```

### Get Connection

```typescript
authlane.connections.get(params: GetConnectionParams): Promise<Response<Connection>>
```

<ParamField path="params.userId" type="string" required>
  User ID
</ParamField>

<ParamField path="params.serviceId" type="string" required>
  Service ID
</ParamField>

```typescript
const { data, error } = await authlane.connections.get({
  userId: 'user_123',
  serviceId: 'github',
});
```

### Get Credentials

```typescript
authlane.connections.getCredentials(params: GetCredentialsParams): Promise<Response<Credentials>>
```

<Warning>
  Credentials are decrypted and returned in plain text. Handle with care!
</Warning>

```typescript
const { data, error } = await authlane.connections.getCredentials({
  userId: 'user_123',
  serviceId: 'github',
});

// data = { access_token: 'ghp_...', refresh_token: '...', expires_at: ... }
```

## Error Handling

The SDK uses Supabase-style error handling:

```typescript
const { data, error } = await authlane.connections.get({
  userId: 'user_123',
  serviceId: 'github',
});

if (error) {
  console.error('Error code:', error.code);
  console.error('Message:', error.message);
  console.error('Hint:', error.hint);
  return;
}

// Use data
console.log(data);
```

### Error Types

```typescript
interface AutoError {
  code: string;
  message: string;
  hint?: string;
  details?: Record<string, any>;
}
```

Common error codes:

- `invalid_api_key` - API key is invalid or revoked
- `connection_not_found` - Connection doesn't exist
- `token_expired` - Access token expired (refresh needed)
- `rate_limit_exceeded` - Too many requests
- `service_unavailable` - External service down

## TypeScript Types

```typescript
interface Connection {
  id: string;
  userId: string;
  serviceId: string;
  tenantId: string;
  status: 'connected' | 'expired' | 'error';
  createdAt: string;
  updatedAt: string;
  lastRefreshedAt?: string;
  expiresAt?: string;
  metadata: Record<string, any>;
}

interface Service {
  id: string;
  name: string;
  category: 'developer_tools' | 'communication' | 'productivity' | 'crm' | 'other';
  authType: 'oauth2' | 'oauth1' | 'api_key';
  isActive: boolean;
  config: {
    scopes?: string[];
    endpoints?: Record<string, string>;
  };
}

interface Credentials {
  access_token: string;
  refresh_token?: string;
  expires_at?: number;
  scope?: string;
  token_type?: string;
  [key: string]: any; // Service-specific fields
}
```

## Advanced Usage

### Custom Headers

```typescript
const authlane = new Authlane({
  apiKey: process.env.AUTHLANE_API_KEY!,
  headers: {
    'X-Custom-Header': 'value',
  },
});
```

### Interceptors

```typescript
authlane.onRequest((config) => {
  console.log('Request:', config.method, config.url);
  return config;
});

authlane.onResponse((response) => {
  console.log('Response:', response.status);
  return response;
});

authlane.onError((error) => {
  console.error('Error:', error.message);
  throw error;
});
```

### Pagination

For endpoints that return lists:

```typescript
let page = 1;
let allConnections: Connection[] = [];

while (true) {
  const { data, error } = await authlane.connections.list({
    userId: 'user_123',
    page,
    perPage: 50,
  });

  if (error) break;
  if (!data || data.length === 0) break;

  allConnections.push(...data);
  page++;
}
```
